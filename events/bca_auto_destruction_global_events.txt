country_event = {
    id = bca_auto_destruction_event.1
    hide_window = yes
    is_triggered_only = yes

    trigger = {
        is_ai = no
    }

    immediate = {
        every_owned_colony = {
            remove_carrier_flag = bca_pf_disable_planet_auto_destruction_building
            remove_carrier_flag = bca_pf_disable_planet_auto_destruction_district
            carrier_event = { id = colony_automation_event.23 }
        }
    }
}
