namespace = bca_auto_destruction_event

country_event = {
    id = bca_auto_destruction_event.1
    hide_window = yes
    is_triggered_only = yes

    trigger = {
        is_ai = no
    }

    immediate = {
        every_owned_planet = {
            limit = {
                is_colony = yes
            }
            remove_planet_flag = bca_pf_disable_planet_auto_destruction_building
            remove_planet_flag = bca_pf_disable_planet_auto_destruction_district
            remove_planet_flag = bca_pf_disable_planet_auto_destruction_zone
        }
    }
}
