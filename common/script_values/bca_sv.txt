bca_potential_district_max_add_energy = {
	set = value:bca_planet_max_districts_of|type|district_generator|

	modifier = {
		bca_can_build_district_hab_energy = yes
		add = value:bca_planet_max_districts_of|type|district_hab_energy|
	}
	modifier = {
		bca_can_build_district_photosynthesis_fields = yes
		add = value:bca_planet_max_districts_of|type|district_photosynthesis_fields|
	}
	modifier = {
		bca_can_build_building_generator_districts_1 = yes
		add = value:bca_potential_energy_district_max_inc
	}
	modifier = {
		OR = {
			bca_can_build_district_geothermal = yes # geothermal district is uncapped
			is_generator_district_uncapped = yes
			bca_has_advance_secondary_districts = yes
		}
		set = value:bca_planet_max_districts
	}
	max = value:bca_planet_max_districts
}
