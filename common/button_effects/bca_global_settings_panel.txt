    potential = { exists = from from = { check_variable = { which = bca_reserve_minerals_amount value > 0 } } }
    allow = { always = yes }
    effect = {
        custom_tooltip = BCA_GLOBAL_SETTINGS_MINERAL_RESERVE_DECREASE_TOOLTIP
        hidden_effect = {
            from = {
                if = {
                    limit = { check_variable = { which = bca_reserve_minerals_amount value <= 2000 } }
                    set_variable = { which = bca_reserve_minerals_amount value = 0 }
                }
                else = {
                    subtract_variable = { which = bca_reserve_minerals_amount value = 2000 }
                }
            }
        }
    }
