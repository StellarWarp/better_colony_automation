bca_refresh_global_settings_panel_counts = {
    set_variable = { which = bca_gs_all_auto_demolition_enabled_planets value = 0 }
    set_variable = { which = bca_gs_all_auto_demolition_disabled_planets value = 0 }
    set_variable = { which = bca_gs_building_auto_demolition_enabled_planets value = 0 }
    set_variable = { which = bca_gs_building_auto_demolition_disabled_planets value = 0 }
    set_variable = { which = bca_gs_district_auto_demolition_enabled_planets value = 0 }
    set_variable = { which = bca_gs_district_auto_demolition_disabled_planets value = 0 }
    set_variable = { which = bca_gs_zone_auto_demolition_enabled_planets value = 0 }
    set_variable = { which = bca_gs_zone_auto_demolition_disabled_planets value = 0 }

    every_owned_colony = {
        if = {
            limit = {
                has_carrier_flag = bca_pf_disable_planet_auto_destruction_building
            }
            owner = {
                add_variable = {
                    which = bca_gs_building_auto_demolition_disabled_planets
                    value = 1
                }
            }
        }
        else = {
            owner = {
                add_variable = {
                    which = bca_gs_building_auto_demolition_enabled_planets
                    value = 1
                }
            }
        }

        if = {
            limit = {
                has_carrier_flag = bca_pf_disable_planet_auto_destruction_district
            }
            owner = {
                add_variable = {
                    which = bca_gs_district_auto_demolition_disabled_planets
                    value = 1
                }
            }
        }
        else = {
            owner = {
                add_variable = {
                    which = bca_gs_district_auto_demolition_enabled_planets
                    value = 1
                }
            }
        }

        if = {
            limit = {
                has_carrier_flag = bca_pf_disable_planet_auto_destruction_zone
            }
            owner = {
                add_variable = {
                    which = bca_gs_zone_auto_demolition_disabled_planets
                    value = 1
                }
            }
        }
        else = {
            owner = {
                add_variable = {
                    which = bca_gs_zone_auto_demolition_enabled_planets
                    value = 1
                }
            }
        }

        if = {
            limit = {
                NOR = {
                    has_carrier_flag = bca_pf_disable_planet_auto_destruction_building
                    has_carrier_flag = bca_pf_disable_planet_auto_destruction_district
                    has_carrier_flag = bca_pf_disable_planet_auto_destruction_zone
                }
            }
            owner = {
                add_variable = {
                    which = bca_gs_all_auto_demolition_enabled_planets
                    value = 1
                }
            }
        }
        if = {
            limit = {
                has_carrier_flag = bca_pf_disable_planet_auto_destruction_building
                has_carrier_flag = bca_pf_disable_planet_auto_destruction_district
                has_carrier_flag = bca_pf_disable_planet_auto_destruction_zone
            }
            owner = {
                add_variable = {
                    which = bca_gs_all_auto_demolition_disabled_planets
                    value = 1
                }
            }
        }
    }
}
