bca_unique_building_potential = {
	bca_can_build_$building1$ = yes
	NOR = {
		has_building = $building1$
		[[building2] has_building = $building2$ ]
		[[building3] has_building = $building3$ ]
		[[building4] has_building = $building4$ ]
	}
}
